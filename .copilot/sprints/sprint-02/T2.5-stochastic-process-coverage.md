```rust
// src/stochastic_process.rs

pub enum DistributionType {
    Uniform { min: f64, max: f64 },
    Normal { mean: f64, std_dev: f64 },
    Discrete { values: Vec<f64>, probabilities: Vec<f64> },
}

pub struct StochasticProcess {
    pub dimension: usize,
    pub stages: Vec<StageProcess>,
}

pub struct StageProcess {
    pub stage: usize,
    pub distributions: Vec<DistributionType>,
}

impl StochasticProcess {
    pub fn new(dimension: usize) -> Self;
    pub fn add_stage(&mut self, stage: StageProcess) -> Result<(), String>;
    pub fn realize(&self, stage: usize, rng: &mut impl Rng) -> Result<Vec<f64>, String>;
    pub fn validate(&self) -> Result<(), String>;
}
```

```rust
// tests/unit_stochastic_process.rs

use powers::stochastic_process::*;
use rand::SeedableRng;
use rand_xoshiro::Xoshiro256Plus;

#[test]
fn test_uniform_distribution_realization() {
    let mut process = StochasticProcess::new(1);

    let stage_process = StageProcess {
        stage: 1,
        distributions: vec![
            DistributionType::Uniform { min: 0.0, max: 10.0 }
        ],
    };

    process.add_stage(stage_process).unwrap();

    let mut rng = Xoshiro256Plus::seed_from_u64(42);
    let realization = process.realize(1, &mut rng).unwrap();

    assert_eq!(realization.len(), 1);
    assert!(realization[0] >= 0.0 && realization[0] <= 10.0);
}

#[test]
fn test_normal_distribution_realization() {
    let mut process = StochasticProcess::new(1);

    let stage_process = StageProcess {
        stage: 1,
        distributions: vec![
            DistributionType::Normal { mean: 100.0, std_dev: 10.0 }
        ],
    };

    process.add_stage(stage_process).unwrap();

    let mut rng = Xoshiro256Plus::seed_from_u64(42);

    // Generate many realizations to test distribution
    let mut realizations = Vec::new();
    for _ in 0..1000 {
        let r = process.realize(1, &mut rng).unwrap();
        realizations.push(r[0]);
    }

    // Check mean is close to expected
    let mean: f64 = realizations.iter().sum::<f64>() / realizations.len() as f64;
    assert!((mean - 100.0).abs() < 5.0, "Mean too far from expected: {}", mean);

    // Check standard deviation is reasonable
    let variance: f64 = realizations.iter()
        .map(|x| (x - mean).powi(2))
        .sum::<f64>() / realizations.len() as f64;
    let std_dev = variance.sqrt();
    assert!((std_dev - 10.0).abs() < 2.0, "Std dev too far from expected: {}", std_dev);
}

#[test]
fn test_discrete_distribution() {
    let mut process = StochasticProcess::new(1);

    let stage_process = StageProcess {
        stage: 1,
        distributions: vec![
            DistributionType::Discrete {
                values: vec![10.0, 20.0, 30.0],
                probabilities: vec![0.2, 0.5, 0.3],
            }
        ],
    };

    process.add_stage(stage_process).unwrap();

    let mut rng = Xoshiro256Plus::seed_from_u64(42);

    // Generate many realizations
    let mut counts = vec![0, 0, 0];
    for _ in 0..1000 {
        let r = process.realize(1, &mut rng).unwrap();

        if (r[0] - 10.0).abs() < 1e-6 {
            counts[0] += 1;
        } else if (r[0] - 20.0).abs() < 1e-6 {
            counts[1] += 1;
        } else if (r[0] - 30.0).abs() < 1e-6 {
            counts[2] += 1;
        } else {
            panic!("Unexpected value: {}", r[0]);
        }
    }

    // Check frequencies match probabilities (approximately)
    assert!((counts[0] as f64 / 1000.0 - 0.2).abs() < 0.05);
    assert!((counts[1] as f64 / 1000.0 - 0.5).abs() < 0.05);
    assert!((counts[2] as f64 / 1000.0 - 0.3).abs() < 0.05);
}
```

```rust
// tests/integration_stochastic_process.rs

use powers::scenario::SAA;
use powers::stochastic_process::*;

#[test]
fn test_process_with_saa() -> Result<(), String> {
    let mut process = StochasticProcess::new(1);

    process.add_stage(StageProcess {
        stage: 1,
        distributions: vec![
            DistributionType::Normal { mean: 100.0, std_dev: 10.0 }
        ],
    })?;

    // Create SAA from process
    let saa = SAA::from_process(&process, 100, 42)?;

    // Verify scenario count
    assert_eq!(saa.scenario_count(1), 100);

    // Verify scenarios are reasonable
    let scenarios = saa.get_scenarios(1)?;
    let mean: f64 = scenarios.iter().sum::<f64>() / scenarios.len() as f64;
    assert!((mean - 100.0).abs() < 5.0);

    Ok(())
}

#[test]
fn test_process_in_sddp() -> Result<(), String> {
    // Create system with stochastic demand
    let mut process = StochasticProcess::new(1);
    process.add_stage(StageProcess {
        stage: 1,

            DistributionType::Uniform { min: 50.0, max: 150.0 }

    let result = sddp.train(10, 5, &saa)?;
```
