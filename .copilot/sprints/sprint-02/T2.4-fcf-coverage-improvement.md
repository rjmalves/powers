use powers::fcf::{Cut, CutPool};

#[test]
fn test_cut_pool_add_and_retrieve() {
    let mut pool = CutPool::new(10);

    let cut = Cut {
        intercept: 100.0,
        coefficients: vec![1.0, 2.0],
        is_active: false,
    };

    pool.add_cut(cut).expect("Should add cut");

    assert_eq!(pool.total_cut_count(), 1);

    let retrieved = pool.get_cut(0).expect("Should retrieve cut");
    assert_eq!(retrieved.intercept, 100.0);
    assert_eq!(retrieved.coefficients, vec![1.0, 2.0]);
}

#[test]
fn test_cut_domination_simple() {
    let mut pool = CutPool::new(10);

    // Add a weak cut
    let weak_cut = Cut {
        intercept: 50.0,
        coefficients: vec![1.0, 1.0],
        is_active: false,
    };
    pool.add_cut(weak_cut).unwrap();

    // Add a stronger cut (dominates the first)
    let strong_cut = Cut {
        intercept: 60.0,  // Higher intercept, same slope
        coefficients: vec![1.0, 1.0],
        is_active: false,
    };
    pool.add_cut(strong_cut).unwrap();

    let dominated = pool.find_dominated_cuts();

    assert_eq!(dominated.len(), 1);
    assert_eq!(dominated[0], 0);  // First cut is dominated
}

#[test]
fn test_cut_pool_capacity() {
    let mut pool = CutPool::new(3);  // Small capacity

    // Add cuts up to capacity
    for i in 0..3 {
        let cut = Cut {
            intercept: (i as f64) * 10.0,
            coefficients: vec![1.0, 1.0],
            is_active: false,
        };
        pool.add_cut(cut).unwrap();
    }

    assert_eq!(pool.total_cut_count(), 3);

    // Add one more cut (exceeds capacity)
    let extra_cut = Cut {
        intercept: 100.0,
        coefficients: vec![2.0, 2.0],
        is_active: false,
    };
    pool.add_cut(extra_cut).unwrap();

    // Pool should still have at most 3 cuts
    // (may have removed oldest/weakest)
    assert!(pool.total_cut_count() <= 3);
}

#[test]
fn test_empty_cut_pool() {
    let pool = CutPool::new(10);

    // Operations on empty pool should not panic
    assert_eq!(pool.total_cut_count(), 0);
    assert_eq!(pool.active_cut_count(), 0);

    let dominated = pool.find_dominated_cuts();
    assert_eq!(dominated.len(), 0);

    // Evaluating at state should return reasonable value
    let value = pool.evaluate_at(&vec![1.0, 2.0]);
    assert!(value.is_finite() || value == f64::NEG_INFINITY);
}

#[test]
fn test_cut_evaluation() {
    let mut pool = CutPool::new(10);

    // Add a cut: future_cost ≥ 100 + 2*x + 3*y
    let cut = Cut {
        intercept: 100.0,
        coefficients: vec![2.0, 3.0],
        is_active: true,
    };
    pool.add_cut(cut).unwrap();

    // Evaluate at state [5, 10]
    let state = vec![5.0, 10.0];
    let value = pool.evaluate_at(&state);

    // Should be 100 + 2*5 + 3*10 = 100 + 10 + 30 = 140
    assert!((value - 140.0).abs() < 1e-6);
}

// tests/integration_fcf_sddp.rs

use powers::sddp::SDDP;

#[test]
fn test_cut_growth_during_training() -> Result<(), String> {
    let system = create_simple_2stage_system();
    let mut sddp = SDDP::new(system, config)?;
    let saa = create_test_saa();

    // Train for a few iterations
    let result = sddp.train(10, 5, &saa)?;

    // Verify cuts were added
    let num_cuts = result.num_cuts;
    assert!(num_cuts > 0, "Should have added cuts");

    // For 2-stage problem, expect roughly num_iterations cuts
    // (may vary with domination)
    assert!(
        num_cuts >= 5 && num_cuts <= 15,
        "Unexpected number of cuts: {}",
        num_cuts
    );

    Ok(())
}

#[test]
fn test_cut_domination_in_training() -> Result<(), String> {
    let system = create_simple_2stage_system();
    let mut config = create_test_config();
    config.enable_cut_domination = true;

    let mut sddp = SDDP::new(system, config)?;
    let saa = create_test_saa();

    // Train long enough for domination to occur
    let result = sddp.train(50, 10, &saa)?;

    // With domination, cut count should stabilize
    // (not grow linearly with iterations)
    let num_cuts = result.num_cuts;
    assert!(
        num_cuts < 50,
        "Domination should limit cut count: {}",
        num_cuts
    );
