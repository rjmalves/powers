### New Test: Monotonicity

```rust
#[test]
fn test_convergence_monotonicity() -> Result<(), String> {
    // Create simple 2-stage problem
    let system = create_simple_2stage_system();
    let config = create_test_config();
    let mut sddp = SDDP::new(system, config)?;
    let saa = create_test_saa();

    // Train with enough iterations to see convergence
    let result = sddp.train(50, 10, &saa)?;

    // Check lower bound monotonicity
    let lower_bounds = result.lower_bounds();
    for (i, window) in lower_bounds.windows(2).enumerate() {
        let prev = window[0];
        let curr = window[1];

        // Allow small numerical error
        assert!(
            curr >= prev - 1e-6,
            "Lower bound decreased at iteration {}: {} -> {}",
            i + 1, prev, curr
        );
    }

    // Check that lower bound actually increased overall
    let initial = lower_bounds[0];
    let final_lb = lower_bounds[lower_bounds.len() - 1];
    assert!(
        final_lb > initial,
        "Lower bound should improve: {} -> {}",
        initial, final_lb
    );

    Ok(())
}
```

```rust
#[test]
fn test_convergence_gap_decrease() -> Result<(), String> {
    let system = create_simple_2stage_system();
    let config = create_test_config();
    let mut sddp = SDDP::new(system, config)?;
    let saa = create_test_saa();

    let result = sddp.train(50, 10, &saa)?;

    // Extract gaps
    let gaps: Vec<f64> = result.iterations()
        .iter()
        .map(|it| it.gap)
        .collect();

    // Initial gap should be larger than final gap
    let initial_gap = gaps[0];
    let final_gap = gaps[gaps.len() - 1];

    assert!(
        final_gap < initial_gap,
        "Gap should decrease: {} -> {}",
        initial_gap, final_gap
    );

    // Final gap should be reasonable
    assert!(
        final_gap < initial_gap * 0.5,
        "Gap should decrease by at least 50%: {} -> {}",
        initial_gap, final_gap
    );

    Ok(())
}
```

```rust
#[test]
fn test_convergence_bounds_validity() -> Result<(), String> {
    let system = create_simple_2stage_system();
    let config = create_test_config();
    let mut sddp = SDDP::new(system, config)?;
    let saa = create_test_saa();

    let result = sddp.train(30, 10, &saa)?;

    // Check every iteration
    for (i, iter_result) in result.iterations().iter().enumerate() {
        // Bounds must be finite
        assert!(
            iter_result.lower_bound.is_finite(),
            "Lower bound is not finite at iteration {}: {}",
            i + 1, iter_result.lower_bound
        );
        assert!(
            iter_result.upper_bound.is_finite(),
            "Upper bound is not finite at iteration {}: {}",
            i + 1, iter_result.upper_bound
        );

        // Lower bound ≤ Upper bound
        assert!(
            iter_result.lower_bound <= iter_result.upper_bound + 1e-6,
            "Lower bound exceeds upper bound at iteration {}: LB={}, UB={}",
            i + 1, iter_result.lower_bound, iter_result.upper_bound
        );

        // Gap should be non-negative (allowing small numerical error)
        assert!(
            iter_result.gap >= -1e-6,
            "Gap is negative at iteration {}: {}",
            i + 1, iter_result.gap
        );
    }

    Ok(())
}
```

### New Test: Stability

```rust
#[test]
fn test_convergence_stability() -> Result<(), String> {
    let system = create_simple_2stage_system();
    let config = create_test_config();
    let mut sddp = SDDP::new(system, config)?;
    let saa = create_test_saa();

    let result = sddp.train(50, 10, &saa)?;

    // Extract gaps
    let gaps: Vec<f64> = result.iterations()
        .iter()
        .map(|it| it.gap)
        .collect();

    // Calculate standard deviation of gaps
    let mean_gap: f64 = gaps.iter().sum::<f64>() / gaps.len() as f64;
    let variance: f64 = gaps.iter()
        .map(|g| (g - mean_gap).powi(2))
        .sum::<f64>() / gaps.len() as f64;
    let std_dev = variance.sqrt();

    // Standard deviation should be less than initial gap
    // (i.e., gaps should stabilize, not oscillate wildly)
    assert!(
        std_dev < gaps[0],
        "Gaps oscillate too much: std_dev={}, initial_gap={}",
        std_dev, gaps[0]
    );

    // Check no large jumps in consecutive gaps
    for (i, window) in gaps.windows(2).enumerate() {
        let prev = window[0];
        let curr = window[1];
        let jump = (curr - prev).abs();

        // Jump should not be larger than 2x the standard deviation
        assert!(
            jump < 2.0 * std_dev,
            "Large jump in gap at iteration {}: {} -> {} (jump: {})",
            i + 1, prev, curr, jump
        );
    }

    Ok(())
}
```

### Test Helper Functions

```rust
// In tests/common/mod.rs or similar

pub fn assert_convergence_quality(result: &TrainingResult) -> Result<(), String> {
    // 1. Check monotonicity
    let lower_bounds = result.lower_bounds();
    for window in lower_bounds.windows(2) {
        if window[1] < window[0] - 1e-6 {
            return Err(format!(
                "Lower bound decreased: {} -> {}",
                window[0], window[1]
            ));
        }
    }

    // 2. Check gap convergence
    let gaps: Vec<f64> = result.iterations()
        .iter()
        .map(|it| it.gap)
        .collect();
    if gaps.last().unwrap() >= gaps.first().unwrap() {
        return Err(format!(
            "Gap did not decrease: {} -> {}",
            gaps.first().unwrap(),
            gaps.last().unwrap()
        ));
    }

    // 3. Check bounds validity
    if !result.final_lower_bound.is_finite() {
        return Err("Final lower bound is not finite".to_string());
    }
    if !result.final_upper_bound.is_finite() {
        return Err("Final upper bound is not finite".to_string());
    }
    if result.final_lower_bound > result.final_upper_bound + 1e-6 {
        return Err(format!(
            "Lower bound exceeds upper bound: {} > {}",
            result.final_lower_bound,
            result.final_upper_bound
        ));
    }

    Ok(())
}

pub fn assert_bounds_in_range(
    result: &TrainingResult,
    expected_min: f64,
    expected_max: f64,
) -> Result<(), String> {
    let lb = result.final_lower_bound;
    let ub = result.final_upper_bound;

    if lb < expected_min || lb > expected_max {
        return Err(format!(
            "Lower bound {} outside expected range [{}, {}]",
            lb, expected_min, expected_max
        ));
    }

    if ub < expected_min || ub > expected_max {
        return Err(format!(
            "Upper bound {} outside expected range [{}, {}]",
            ub, expected_min, expected_max
        ));
    }

    Ok(())
}

pub fn print_convergence_summary(result: &TrainingResult) {
    println!("\n=== Convergence Summary ===");
    println!("Total iterations: {}", result.iterations().len());
    println!("Final lower bound: {:.4}", result.final_lower_bound);
    println!("Final upper bound: {:.4}", result.final_upper_bound);
    println!("Final gap: {:.4}", result.final_gap());
    println!("Relative gap: {:.4}%", result.relative_gap() * 100.0);
    println!("Best upper bound: {:.4} (iteration {})",
             result.best_upper_bound, result.best_iteration);
    println!("Total time: {:?}", result.total_time);
    println!("===========================\n");
}
```

## Testing Strategy

### Test Matrix

| Test Type       | What It Validates           | Tolerance          |
| --------------- | --------------------------- | ------------------ |
| Monotonicity    | Lower bounds non-decreasing | 1e-6               |
| Gap Decrease    | Gap reduces over time       | Initial gap \* 0.5 |
| Bounds Validity | LB ≤ UB, finite             | 1e-6               |
| Stability       | No wild oscillations        | 2x std dev         |
| Range Check     | Bounds in expected range    | Problem-specific   |

### Tolerance Levels

**Why 1e-6 for monotonicity?**

- Numerical precision in MILP solvers
- Small floating-point errors acceptable
- Stricter would cause false failures

**Why 50% gap reduction?**

- Conservative target for general problems
- Some problems converge slower
- Avoids brittle tests

**Why 2x std_dev for stability?**

- Statistical significance
- Allows normal variance
- Catches pathological behavior
