### Mock Solver

```rust
// tests/common/mock_solver.rs

use powers::solver::{Solver, SolverResult, SolverStatus};

pub struct MockSolver {
    pub should_fail: bool,
    pub status: SolverStatus,
    pub objective_value: f64,
    pub solution: Vec<f64>,
}

impl MockSolver {
    pub fn new_success(obj: f64, solution: Vec<f64>) -> Self {
        Self {
            should_fail: false,
            status: SolverStatus::Optimal,
            objective_value: obj,
            solution,
        }
    }

    pub fn new_infeasible() -> Self {
        Self {
            should_fail: true,
            status: SolverStatus::Infeasible,
            objective_value: f64::NAN,
            solution: vec![],
        }
    }

    pub fn new_unbounded() -> Self {
        Self {
            should_fail: true,
            status: SolverStatus::Unbounded,
            objective_value: f64::NEG_INFINITY,
            solution: vec![],
        }
    }
}

impl Solver for MockSolver {
    fn solve(&mut self, problem: &Problem) -> Result<SolverResult, String> {
        if self.should_fail {
            return Err(format!("Solver failed: {:?}", self.status));
        }

        Ok(SolverResult {
            status: self.status.clone(),
            objective_value: self.objective_value,
            solution: self.solution.clone(),
            solve_time: std::time::Duration::from_millis(1),
        })
    }
}
```

```rust
#[test]
fn test_real_solver_simple_lp() -> Result<(), String> {
    // Minimize: x + 2y
    // Subject to: x + y >= 1
    //             x, y >= 0
    // Optimal: x=1, y=0, obj=1

    let mut problem = Problem::new();
    problem.add_variable("x", 0.0, f64::INFINITY, 1.0);  // cost 1
    problem.add_variable("y", 0.0, f64::INFINITY, 2.0);  // cost 2
    problem.add_constraint("demand", vec![1.0, 1.0], 1.0, f64::INFINITY);  // x + y >= 1

    let mut solver = HiGHSSolver::new()?;
    let result = solver.solve(&problem)?;

    assert_eq!(result.status, SolverStatus::Optimal);
    assert!((result.objective_value - 1.0).abs() < 1e-6);
    assert!((result.solution[0] - 1.0).abs() < 1e-6);  // x = 1
    assert!(result.solution[1].abs() < 1e-6);  // y = 0

    Ok(())
}

#[test]
fn test_real_solver_infeasible() -> Result<(), String> {
    // Infeasible problem:
    // x >= 10
    // x <= 5

    let mut problem = Problem::new();
    problem.add_variable("x", 0.0, f64::INFINITY, 1.0);
    problem.add_constraint("lower", vec![1.0], 10.0, f64::INFINITY);  // x >= 10
    problem.add_constraint("upper", vec![1.0], f64::NEG_INFINITY, 5.0);  // x <= 5

    let mut solver = HiGHSSolver::new()?;
    let result = solver.solve(&problem);

    // Should detect infeasibility
    assert!(result.is_err() || result.unwrap().status == SolverStatus::Infeasible);

    Ok(())
}

#[test]
fn test_real_solver_unbounded() -> Result<(), String> {
    // Unbounded problem:
    // Minimize: -x (no upper bound on x)

    let mut problem = Problem::new();
    problem.add_variable("x", 0.0, f64::INFINITY, -1.0);  // Minimize -x
    // No constraints

    let mut solver = HiGHSSolver::new()?;
    let result = solver.solve(&problem);

    // Should detect unboundedness
    assert!(result.is_err() || result.unwrap().status == SolverStatus::Unbounded);

    Ok(())
}
```

### Test: Performance

```rust
#[test]
fn test_solver_performance() -> Result<(), String> {
    // Create problem with 1000 variables and 500 constraints
    let mut problem = Problem::new();

    for i in 0..1000 {
        problem.add_variable(&format!("x{}", i), 0.0, 1.0, 1.0);
    }

    for i in 0..500 {
        let coeffs: Vec<f64> = (0..1000).map(|_| 1.0).collect();
        problem.add_constraint(&format!("c{}", i), coeffs, 0.0, 100.0);
    }

    let mut solver = HiGHSSolver::new()?;

    let start = std::time::Instant::now();
    let result = solver.solve(&problem)?;
    let elapsed = start.elapsed();

    assert_eq!(result.status, SolverStatus::Optimal);

    // Should solve in reasonable time (< 1 second for this size)
    assert!(
        elapsed < std::time::Duration::from_secs(1),
        "Solve took too long: {:?}",
        elapsed
    );

    println!("Solved 1000x500 problem in {:?}", elapsed);

    Ok(())
}

#[test]
fn test_solver_repeated_solves() -> Result<(), String> {
    let problem = create_standard_test_problem();
    let mut solver = HiGHSSolver::new()?;

    let mut solve_times = Vec::new();

    for _ in 0..100 {
        let start = std::time::Instant::now();
        let result = solver.solve(&problem)?;
        let elapsed = start.elapsed();

        assert_eq!(result.status, SolverStatus::Optimal);
        solve_times.push(elapsed);
    }

    // Check mean and variance
    let mean: std::time::Duration = solve_times.iter().sum::<std::time::Duration>() / 100;

    println!("Mean solve time over 100 solves: {:?}", mean);

    // Should be consistent (not increasing - no memory leak)
    let first_10 = &solve_times[0..10];
    let last_10 = &solve_times[90..100];
    let mean_first: std::time::Duration = first_10.iter().sum::<std::time::Duration>() / 10;
    let mean_last: std::time::Duration = last_10.iter().sum::<std::time::Duration>() / 10;

    assert!(
        mean_last < mean_first * 2,
        "Solve time increased significantly: {:?} -> {:?}",
        mean_first, mean_last
    );

    Ok(())
}
```

### Test: Thread Safety

```rust
#[test]
fn test_solver_thread_safety() -> Result<(), String> {
    use std::sync::{Arc, Mutex};
    use std::thread;

    let problem = Arc::new(create_standard_test_problem());
    let mut handles = vec![];

    for i in 0..10 {
        let problem_clone = Arc::clone(&problem);

        let handle = thread::spawn(move || {
            let mut solver = HiGHSSolver::new().unwrap();
            let result = solver.solve(&problem_clone).unwrap();

            assert_eq!(result.status, SolverStatus::Optimal);

            println!("Thread {} solved successfully", i);
        });

        handles.push(handle);
    }

    for handle in handles {
        handle.join().unwrap();
    }

    Ok(())
}
```
