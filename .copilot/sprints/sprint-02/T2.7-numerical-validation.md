## Technical Implementation

### Test: Convergence to Known Solution

```rust
// tests/integration_numerical_validation.rs

use tests::benchmarks::*;
use powers::sddp::SDDP;

#[test]
fn test_convergence_to_newsvendor_solution() -> Result<(), String> {
    let (system, config, saa, solution) = create_newsvendor_benchmark();

    let mut sddp = SDDP::new(system, config)?;
    let result = sddp.train(100, 30, &saa)?;

    // 1. Check convergence
    assert!(
        result.converged(0.02),  // 2% relative gap
        "Failed to converge: gap = {:.4}",
        result.relative_gap()
    );

    // 2. Validate bounds are near known solution
    let lb = result.final_lower_bound;
    let ub = result.final_upper_bound;
    let expected = solution.optimal_value;
    let tol = solution.tolerance;

    assert!(
        lb >= expected - tol,
        "Lower bound too low: {} < {} - {} (error: {})",
        lb, expected, tol, expected - lb
    );

    assert!(
        ub <= expected + tol,
        "Upper bound too high: {} > {} + {} (error: {})",
        ub, expected, tol, ub - expected
    );

    // 3. Check convergence rate
    let gaps: Vec<f64> = result.iterations()
        .iter()
        .map(|it| it.gap)
        .collect();

    let initial_gap = gaps[0];
    let final_gap = gaps[gaps.len() - 1];

    assert!(
        final_gap < 0.5 * initial_gap,
        "Gap should reduce by at least 50%: {} -> {}",
        initial_gap, final_gap
    );

    println!("✓ Newsvendor: LB={:.2}, UB={:.2}, Expected={:.2}, Gap={:.2}%",
             lb, ub, expected, result.relative_gap() * 100.0);

    Ok(())
}
```

### Test: Bound Properties

```rust
#[test]
fn test_bound_validity() -> Result<(), String> {
    let (system, config, saa, _) = create_newsvendor_benchmark();

    let mut sddp = SDDP::new(system, config)?;
    let result = sddp.train(50, 20, &saa)?;

    // Check at every iteration
    for (i, iter) in result.iterations().iter().enumerate() {
        // Lower bound ≤ Upper bound (fundamental property)
        assert!(
            iter.lower_bound <= iter.upper_bound + 1e-6,
            "Iteration {}: LB > UB ({} > {})",
            i + 1, iter.lower_bound, iter.upper_bound
        );

        // Bounds are finite
        assert!(
            iter.lower_bound.is_finite(),
            "Iteration {}: LB is not finite: {}",
            i + 1, iter.lower_bound
        );

        assert!(
            iter.upper_bound.is_finite(),
            "Iteration {}: UB is not finite: {}",
            i + 1, iter.upper_bound
        );

        // Gap is non-negative
        assert!(
            iter.gap >= -1e-6,
            "Iteration {}: Negative gap: {}",
            i + 1, iter.gap
        );
    }

    Ok(())
}

#[test]
fn test_lower_bound_monotonicity() -> Result<(), String> {
    let (system, config, saa, _) = create_newsvendor_benchmark();

    let mut sddp = SDDP::new(system, config)?;
    let result = sddp.train(50, 20, &saa)?;

    let lower_bounds = result.lower_bounds();

    for (i, window) in lower_bounds.windows(2).enumerate() {
        let prev = window[0];
        let curr = window[1];

        assert!(
            curr >= prev - 1e-6,
            "Lower bound decreased at iteration {}: {} -> {} (decrease: {})",
            i + 2, prev, curr, prev - curr
        );
    }

    // Overall improvement
    assert!(
        lower_bounds.last().unwrap() > lower_bounds.first().unwrap(),
        "Lower bound should improve overall"
    );

    Ok(())
}
```

### Test: Stability Across Seeds

```rust
#[test]
fn test_stability_across_seeds() -> Result<(), String> {
    let seeds = vec![42, 123, 456, 789, 1011];
    let mut final_bounds = Vec::new();

    for seed in seeds {
        let (mut system, config, _, solution) = create_newsvendor_benchmark();
        system.seed = seed;

        let saa = SAA::from_process(&system.process, 100, seed)?;

        let mut sddp = SDDP::new(system, config)?;
        let result = sddp.train(50, 20, &saa)?;

        final_bounds.push((result.final_lower_bound, result.final_upper_bound));
    }

    // Calculate mean and standard deviation of bounds
    let mean_lb: f64 = final_bounds.iter().map(|(lb, _)| lb).sum::<f64>() / seeds.len() as f64;
    let mean_ub: f64 = final_bounds.iter().map(|(_, ub)| ub).sum::<f64>() / seeds.len() as f64;

    let var_lb: f64 = final_bounds.iter()
        .map(|(lb, _)| (lb - mean_lb).powi(2))
        .sum::<f64>() / seeds.len() as f64;
    let std_lb = var_lb.sqrt();

    let var_ub: f64 = final_bounds.iter()
        .map(|(_, ub)| (ub - mean_ub).powi(2))
        .sum::<f64>() / seeds.len() as f64;
    let std_ub = var_ub.sqrt();

    // Standard deviation should be small relative to mean
    assert!(
        std_lb < 0.1 * mean_lb.abs(),
        "Lower bound too unstable across seeds: mean={}, std={}",
        mean_lb, std_lb
    );

    assert!(
        std_ub < 0.1 * mean_ub.abs(),
        "Upper bound too unstable across seeds: mean={}, std={}",
        mean_ub, std_ub
    );

    println!("Stability: LB: {:.2} ± {:.2}, UB: {:.2} ± {:.2}",
             mean_lb, std_lb, mean_ub, std_ub);

    Ok(())
}
```

### Test: Robustness to Parameters

```rust
#[test]
fn test_robustness_to_iterations() -> Result<(), String> {
    let iteration_counts = vec![10, 30, 50, 100];
    let mut gaps = Vec::new();

    for num_iter in iteration_counts {
        let (system, mut config, saa, _) = create_newsvendor_benchmark();
        config.num_iterations = num_iter;

        let mut sddp = SDDP::new(system, config)?;
        let result = sddp.train(num_iter, 20, &saa)?;

        gaps.push(result.final_gap());
    }

    // More iterations → smaller gap (or at least not worse)
    for window in gaps.windows(2) {
        assert!(
            window[1] <= window[0] + 10.0,  // Allow some variance
            "More iterations should not worsen gap: {} -> {}",
            window[0], window[1]
        );
    }

    println!("Gaps vs iterations: {:?}", gaps);

    Ok(())
}

#[test]
fn test_robustness_to_forward_passes() -> Result<(), String> {
    let forward_counts = vec![5, 10, 20, 50];
    let mut ub_variances = Vec::new();

    for num_forward in forward_counts {
        let (system, config, saa, _) = create_newsvendor_benchmark();

        let mut sddp = SDDP::new(system, config)?;
        let result = sddp.train(30, num_forward, &saa)?;

        // Compute variance of upper bounds
        let upper_bounds = result.upper_bounds();
        let mean_ub: f64 = upper_bounds.iter().sum::<f64>() / upper_bounds.len() as f64;
        let variance: f64 = upper_bounds.iter()
            .map(|ub| (ub - mean_ub).powi(2))
            .sum::<f64>() / upper_bounds.len() as f64;

        ub_variances.push(variance);
    }

    // More forward passes → lower variance (more stable UB)
    // (May not be perfectly monotonic due to randomness)
    assert!(
        ub_variances.last().unwrap() < ub_variances.first().unwrap(),
        "More forward passes should reduce UB variance"
    );

    println!("UB variance vs forward passes: {:?}", ub_variances);

    Ok(())
}
```

### Test: Edge Cases

```rust
#[test]
fn test_deterministic_problem_convergence() -> Result<(), String> {
    // Create newsvendor with deterministic demand (single value)
    let (mut system, config, _, _) = create_newsvendor_benchmark();

    // Replace process with deterministic
    let mut process = StochasticProcess::new(1);
    process.add_stage(StageProcess {
        stage: 2,
        distributions: vec![
            DistributionType::Discrete {
                values: vec![50.0],
                probabilities: vec![1.0],
            }
        ],
    }).unwrap();

    system.process = process;
    let saa = SAA::from_process(&system.process, 1, 42)?;

    let mut sddp = SDDP::new(system, config)?;
    let result = sddp.train(10, 1, &saa)?;

    // Deterministic problem: should converge quickly, LB ≈ UB
    assert!(
        result.final_gap() < 1.0,
        "Deterministic problem should have very small gap: {}",
        result.final_gap()
    );

    Ok(())
}

#[test]
fn test_high_variance_problem() -> Result<(), String> {
    let (mut system, config, _, _) = create_newsvendor_benchmark();

    // Replace with high variance demand
    let mut process = StochasticProcess::new(1);
    process.add_stage(StageProcess {
        stage: 2,
        distributions: vec![
            DistributionType::Normal { mean: 50.0, std_dev: 50.0 }  // Large variance
        ],
    }).unwrap();

    system.process = process;
    let saa = SAA::from_process(&system.process, 200, 42)?;  // More scenarios

    let mut sddp = SDDP::new(system, config)?;
    let result = sddp.train(100, 30, &saa)?;

    // High variance: convergence may be slower but should still work
    assert!(
        result.converged(0.1),  // More lenient tolerance
        "High variance problem should still converge: gap={}",
        result.relative_gap()
    );

    Ok(())
}
```
