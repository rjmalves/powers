### Test Naming Convention

```rust
#[test]
fn test_<workflow>_<scenario>_<expected_outcome>() {
    // Example: test_training_empty_scenarios_returns_error
}
```

### Example Integration Test

```rust
#[test]
fn test_training_converges_on_small_hydrothermal() {
    // Arrange
    let config = load_config("tests/fixtures/small_hydrothermal/config.json");
    let system = load_system("tests/fixtures/small_hydrothermal/system.json");
    let graph = load_graph("tests/fixtures/small_hydrothermal/graph.json");
    let scenarios = load_scenarios("tests/fixtures/small_hydrothermal/scenario.json");

    let mut sddp = SDDP::new(config, system, graph);

    // Act
    let result = sddp.train(&scenarios, 100);

    // Assert
    assert!(result.is_ok());
    let stats = result.unwrap();
    assert!(stats.converged);
    assert!(stats.iterations < 100);
    assert!(stats.upper_bound - stats.lower_bound < 100.0);  // Convergence gap
    assert!(stats.policy_cost > 0.0);
}
```

### Edge Case Test Pattern

```rust
#[test]
fn test_training_single_stage_problem() {
    // Single-stage problem should not add cuts (no future)
    let mut sddp = create_single_stage_problem();
    let scenarios = create_test_scenarios(10);

    let result = sddp.train(&scenarios, 10);

    assert!(result.is_ok());
    assert_eq!(sddp.total_cuts(), 0);  // No cuts for single stage
}

#[test]
fn test_training_empty_scenarios() {
    let mut sddp = create_test_problem();
    let scenarios = vec![];  // Empty

    let result = sddp.train(&scenarios, 10);

    // Should handle gracefully (no-op or error)
    assert!(result.is_ok() || result.is_err());
    if let Err(e) = result {
        assert_eq!(e.kind(), ErrorKind::InvalidInput);
    }
}
```

### Error Handling Test Pattern

```rust
#[test]
fn test_infeasible_subproblem_returns_error() {
    // Create problem with conflicting constraints
    let config = create_infeasible_config();
    let mut sddp = SDDP::from_config(&config);

    let result = sddp.train(&scenarios, 10);

    assert!(result.is_err());
    assert_eq!(result.unwrap_err().kind(), ErrorKind::InfeasibleProblem);
}

#[test]
fn test_invalid_json_returns_error() {
    let result = load_config("tests/fixtures/invalid.json");

    assert!(result.is_err());
    assert_eq!(result.unwrap_err().kind(), ErrorKind::ParseError);
}
```

### Realistic Problem Test Pattern

```rust
#[test]
#[ignore]  // Long-running test, run with --ignored
fn test_training_24_stage_hourly_problem() {
    let config = load_config("tests/fixtures/24stage_hourly/config.json");
    let system = load_system("tests/fixtures/24stage_hourly/system.json");
    let graph = load_graph("tests/fixtures/24stage_hourly/graph.json");
    let scenarios = load_scenarios("tests/fixtures/24stage_hourly/scenario.json");

    let mut sddp = SDDP::new(config, system, graph);

    let start = Instant::now();
    let result = sddp.train(&scenarios, 100);
    let duration = start.elapsed();

    assert!(result.is_ok());
    assert!(duration < Duration::from_secs(300));  // <5 min
}
```

### Test Helpers

```rust
// tests/common/problem_builders.rs

pub fn create_small_hydrothermal() -> SDDP {
    let config = SDDPConfig {
        max_iterations: 50,
        convergence_tolerance: 1e-3,
        ..Default::default()
    };

    let system = System {
        generators: vec![
            Generator::hydro("hydro1", 100.0, 500.0),
            Generator::thermal("thermal1", 50.0, 200.0),
        ],
        demands: vec![Demand { stage: 0, value: 120.0 }],
    };

    let graph = Graph::two_stage();

    SDDP::new(config, system, graph)
}

pub fn create_test_scenarios(count: usize) -> Vec<Scenario> {
    (0..count)
        .map(|i| Scenario {
            inflows: vec![100.0 + (i as f64) * 10.0],
            demands: vec![120.0],
        })
        .collect()
}
```
